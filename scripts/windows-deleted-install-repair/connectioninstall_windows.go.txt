//go:build windows && amd64

// The operator kit delegates legacy cleanup to the same production verifier.
package connectioninstall

import (
	"encoding/json"
	"fmt"
	"golang.org/x/sys/windows"
	"golang.org/x/sys/windows/registry"
	"golang.org/x/sys/windows/svc"
	"golang.org/x/sys/windows/svc/mgr"
	"path/filepath"
	"reflect"
	"strings"
)

func ObserveReinstalledCore(expectedSHA, expectedVersion string) error {
	root, e := Root()
	if e != nil {
		return e
	}
	exe := filepath.Join(root, CurrentApplicationDirectory, BinaryName)
	if e = ownedRegistration(); e != nil {
		return e
	}
	k, e := registry.OpenKey(registry.LOCAL_MACHINE, uninstallKey, registry.READ)
	if e != nil {
		return e
	}
	defer k.Close()
	v, _, e := k.GetStringValue("DisplayVersion")
	if e != nil || v != expectedVersion {
		return fmt.Errorf("REINSTALL_REGISTRATION_NOT_READY")
	}
	for name, want := range map[string]string{"DisplayIcon": exe, "InstallLocation": filepath.Dir(exe)} {
		v, _, err := k.GetStringValue(name)
		if err != nil || v != want {
			return fmt.Errorf("REINSTALL_REGISTRATION_ACTION_CHANGED")
		}
	}
	command, _, e := k.GetStringValue("UninstallString")
	args, parse := windows.DecomposeCommandLine(command)
	if e != nil || parse != nil || !reflect.DeepEqual(args, []string{exe, "--uninstall"}) {
		return fmt.Errorf("REINSTALL_REGISTRATION_ACTION_CHANGED")
	}
	m, e := mgr.Connect()
	if e != nil {
		return e
	}
	defer m.Disconnect()
	s, binding, e := ownedCoreService(m)
	if e != nil {
		return e
	}
	defer s.Close()
	if !strings.EqualFold(binding.Executable, exe) || len(expectedSHA) != 64 || binding.SHA256 != expectedSHA {
		return fmt.Errorf("REINSTALL_CORE_VERSION_CHANGED")
	}
	status, e := s.Query()
	if e != nil || status.State != svc.Running || status.ProcessId == 0 {
		return fmt.Errorf("REINSTALL_CORE_NOT_RUNNING")
	}
	t, present, e := readInstallationTransaction()
	if e != nil || !present || t.Phase != InstallComplete || t.Next.Version != expectedVersion || t.Next.SHA != expectedSHA {
		return fmt.Errorf("REINSTALL_TRANSACTION_NOT_ACCEPTED")
	}
	state, e := StateDirectory()
	if e != nil {
		return e
	}
	raw, e := coreProtectedRead(filepath.Join(state, installationReadyName), false, true)
	var ready installationReady
	if e != nil || json.Unmarshal(raw, &ready) != nil || ready.SHA256 != expectedSHA || !installationReadyProcess(ready) {
		return fmt.Errorf("REINSTALL_INITIALIZATION_NOT_ACCEPTED")
	}
	return nil
}
